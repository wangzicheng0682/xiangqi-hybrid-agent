### 1.3 静态训练 vs 动态检测

| 维度 | 静态标签（训练） | 动态标签（运行时） |
|------|-----------------|-------------------|
| **数据来源** | 当前局面状态 | 遍历合法走法后检测 |
| **训练方式** | Transformer拟合 | 规则引擎硬检 |
| **准确性** | 拟合率（可能<100%） | 100%（规则保证） |
| **示例** | is_check, is_pinned | is_discovered_check, is_double_attack |
| **用途** | 快速预测+辅助训练 | 生成完整Evidence Map |

## 二、战术标签检测器设计

### 2.1 六大层级标签体系

```
层级1: 将杀困毙层（全局・最高优先级）
├── is_check        - 被将军
├── is_checkmate    - 将死
└── is_stalemate    - 困毙

层级2: 捉子层（全局/局部）
├── is_attack_unprotected  - 攻无根
├── is_mutual_attack       - 互吃
└── is_attack_king_adjacent - 攻将邻

层级3: 闪击抽将层（全局・战术型）
├── is_discovered_check        - 闪将
├── is_double_attack_with_check - 抽将
└── is_discovered_attack       - 闪击

层级4: 牵制串打层（全局・战术型）
├── is_pinned             - 被牵制
└── is_cannon_double_attack - 炮串打

层级5: 标准杀法层（杀法型）
├── checkmate_horse_back_cannon - 马后炮
├── checkmate_iron_gate         - 铁门栓
└── checkmate_double_rook       - 双车错

层级6: 子力状态层（单棋子属性）
├── piece_is_unprotected - 子无根
└── cannon_has_platform  - 炮有架
```

### 2.2 检测器接口设计

```python
from dataclasses import dataclass
from typing import List, Dict, Any, Optional
from enum import Enum

class DetectionMode(Enum):
    """检测模式"""
    STATIC = "static"      # 静态检测：仅当前局面
    DYNAMIC = "dynamic"    # 动态检测：遍历合法走法
    FULL = "full"          # 完整检测：静态+动态

@dataclass
class TacticalTagResult:
    """单个标签检测结果"""
    tag_name: str                    # 标签名（如 "is_check"）
    detected: bool                   # 是否检测到
    confidence: float                # 置信度（0.0或1.0，规则硬检）
    bind_pieces: List[str]           # 绑定棋子 ["红车-(5,8)", ...]
    metadata: Optional[Dict] = None  # 额外元数据

@dataclass
class StaticDetectionResult:
    """静态检测结果"""
    fen: str                                # 输入FEN
    tags: List[TacticalTagResult]           # 所有标签结果
    detected_tags: List[TacticalTagResult]  # 仅检测到的标签

@dataclass
class DynamicDetectionResult:
    """动态检测结果（针对单步走法）"""
    move: str                               # UCI走法
    tags: List[TacticalTagResult]           # 该走法产生的战术标签

@dataclass
class FullDetectionResult:
    """完整检测结果"""
    fen: str                                      # 输入FEN
    static_result: StaticDetectionResult          # 静态检测结果
    dynamic_results: Dict[str, DynamicDetectionResult]  # 走法→动态结果
    
    def to_evidence_map(self) -> Dict[str, Any]:
        """转换为Evidence Map格式"""
        return {
            "facts_rules": [
                {
                    "tag": t.tag_name,
                    "confidence": t.confidence,
                    "bind_pieces": t.bind_pieces,
                    "source": "static"
                }
                for t in self.static_result.detected_tags
            ] + [
                {
                    "tag": t.tag_name,
                    "confidence": t.confidence,
                    "bind_pieces": t.bind_pieces,
                    "source": "dynamic",
                    "move": move
                }
                for move, dyn in self.dynamic_results.items()
                for t in dyn.tags if t.detected
            ]
        }

            fen: 当前局面FEN
            
        Returns:
            StaticDetectionResult: 静态检测结果
        """
        pass
    
    def detect_dynamic(self, fen: str, move: str) -> DynamicDetectionResult:
        """
        动态检测：分析某步棋产生的战术效果
        
        Args:
            fen: 当前局面FEN
            move: UCI格式走法
            
        Returns:
            DynamicDetectionResult: 动态检测结果

        pass
    
    def detect_full(self, fen: str,
```

### 2.3 静态检测算法

#### 2.3.1 第一层：将杀困毙检测

```python
def detect_check_mate_stalemate(self, board: List[List[str]], 
                                 is_red_turn: bool) -> List[TacticalTagResult]:
    """
    检测将杀困毙层标签
    
    算法：
    1. 调用规则引擎获取合法走法
    2. 检测是否被将军
    3. 根据合法走法数量判断将死/困毙
    """
    results = []
    
    # 获取合法走法
    legal_moves = self.rule_engine.get_legal_moves(board)
    is_in_check = self.rule_engine.is_in_check(board, is_red_turn)
    
    # 检测 is_check
    if is_in_check:
        attacker = self._find_check_attacker(board, is_red_turn)
        king_pos = self._find_king(board, is_red_turn)
        results.append(TacticalTagResult(
            tag_name="is_check",
            detected=True,
            confidence=1.0,
            bind_pieces=[
                self._bind_piece(attacker),
                self._bind_piece_at(king_pos, is_red_turn, "将")
            ]
        ))
    
    # 检测 is_checkmate
    if is_in_check and len(legal_moves) == 0:
        results.append(TacticalTagResult(
            tag_name="is_checkmate",
            detected=True,
            confidence=1.0,
            bind_pieces=[self._bind_piece_at(king_pos, is_red_turn, "将")]
        ))
    
    # 检测 is_stalemate
    if not is_in_check and len(legal_moves) == 0:
        results.append(TacticalTagResult(
            tag_name="is_stalemate",
            detected=True,
            confidence=1.0,
            bind_pieces=[self._bind_piece_at(king_pos, is_red_turn, "将")]
        ))
    
    return results
```

#### 2.3.2 第二层：捉子检测

```python
def detect_capture(self, board: List[List[str]], 
                   is_red_turn: bool) -> List[TacticalTagResult]:
    """
    检测捉子层标签
    
    算法：
    1. 遍历己方所有棋子的攻击范围
    2. 检测攻击目标是否无保护
    3. 检测是否存在互吃关系
    """
    results = []
    
    # 获取当前方所有棋子
    my_pieces = self._get_my_pieces(board, is_red_turn)
    enemy_pieces = self._get_enemy_pieces(board, is_red_turn)
    
    for piece_pos in my_pieces:
        attacks = self._get_attacks(board, piece_pos)
        
        for target_pos in attacks:
            target_piece = board[target_pos[0]][target_pos[1]]
            if not target_piece:
                continue
            
            # 检测 is_attack_unprotected
            if not self._is_protected(board, target_pos, not is_red_turn):
                results.append(TacticalTagResult(
                    tag_name="is_attack_unprotected",
                    detected=True,
                    confidence=1.0,
                    bind_pieces=[
                        self._bind_piece_at(piece_pos, is_red_turn),
                        self._bind_piece_at(target_pos, not is_red_turn)
                    ]
                ))
            
            # 检测 is_mutual_attack
            if self._can_attack(board, target_pos, piece_pos):
                results.append(TacticalTagResult(
                    tag_name="is_mutual_attack",
                    detected=True,
                    confidence=1.0,
                    bind_pieces=[
                        self._bind_piece_at(piece_pos, is_red_turn),
                        self._bind_piece_at(target_pos, not is_red_turn)
                    ]
                ))
    
    return results
```

#### 2.3.3 第三层：闪击抽将检测（动态）

```python
def detect_discovered_attack(self, board: List[List[str]], 
                              move: str, 
                              is_red_turn: bool) -> List[TacticalTagResult]:
    """
    检测闪击抽将层标签（动态检测）
    
    算法：
    1. 执行走法前，记录被移动棋子后方的远程棋子
    2. 执行走法后，检测该远程棋子是否形成新的攻击
    3. 判断是否为将军或攻击无根子
    """
    results = []
    
    # 解析走法
    from_pos, to_pos = self._parse_move(move)
    moving_piece = board[from_pos[0]][from_pos[1]]
    
    # 找到移动棋子后方的远程棋子（车/炮）
    back_attacker = self._find_back_attacker(board, from_pos, is_red_turn)
    if not back_attacker:
        return results
    
    # 执行走法
    new_board = self._apply_move(board, move)
    
    # 检测移动后是否形成将军
    if self.rule_engine.is_in_check(new_board, not is_red_turn):
        results.append(TacticalTagResult(
            tag_name="is_discovered_check",
            detected=True,
            confidence=1.0,
            bind_pieces=[
                self._bind_piece_at(from_pos, is_red_turn),
                self._bind_piece_at(back_attacker, is_red_turn)
            ],
            metadata={"move": move}
        ))
    
    # 检测移动后是否攻击无根子
    new_attacks = self._get_attacks(new_board, back_attacker)
    for target_pos in new_attacks:
        if not self._is_protected(new_board, target_pos, not is_red_turn):
            results.append(TacticalTagResult(
                tag_name="is_discovered_attack",
                detected=True,
                confidence=1.0,
                bind_pieces=[
                    self._bind_piece_at(from_pos, is_red_turn),
                    self._bind_piece_at(back_attacker, is_red_turn),
                    self._bind_piece_at(target_pos, not is_red_turn)
                ],
                metadata={"move": move}
            ))
    
    return results


def detect_double_attack_with_check(self, board: List[List[str]],
                                     move: str,
                                     is_red_turn: bool) -> List[TacticalTagResult]:
    """
    检测抽将标签（动态检测）
    
    算法：
    1. 执行走法后检测是否将军
    2. 同时检测是否攻击无根子或关键子
    """
    results = []
    
    # 执行走法
    new_board = self._apply_move(board, move)
    
    # 检测是否将军
    if not self.rule_engine.is_in_check(new_board, not is_red_turn):
        return results
    
    # 找到执行走法的棋子
    from_pos, to_pos = self._parse_move(move)
    attacker_pos = to_pos
    
    # 检测该棋子是否同时攻击其他目标
    attacks = self._get_attacks(new_board, attacker_pos)
    for target_pos in attacks:
        target_piece = new_board[target_pos[0]][target_pos[1]]
        if target_piece and self._is_enemy(target_piece, is_red_turn):
            # 攻击敌方棋子 + 将军 = 抽将
            results.append(TacticalTagResult(
                tag_name="is_double_attack_with_check",
                detected=True,
                confidence=1.0,
                bind_pieces=[
                    self._bind_piece_at(attacker_pos, is_red_turn),
                    self._bind_piece_at(target_pos, not is_red_turn)
                ],
                metadata={"move": move}
            ))
    
    return results
```

#### 2.3.4 第四层：牵制串打检测

```python
def detect_pin(self, board: List[List[str]], 
               is_red_turn: bool) -> List[TacticalTagResult]:
    """
    检测牵制标签
    
    算法：
    1. 找到敌方车/炮
    2. 检测是否存在己方棋子在车/炮与将/帅的连线上
    3. 验证移动该棋子会导致被将军
    """
    results = []
    
    king_pos = self._find_king(board, is_red_turn)
    enemy_pinners = self._get_enemy_pinners(board, is_red_turn)  # 车/炮
    
    for pinner_pos in enemy_pinners:
        # 检测与将/帅的连线
        line = self._get_line(pinner_pos, king_pos)
        if not line:
            continue
        
        # 找到连线上的己方棋子
        for pos in line:
            piece = board[pos[0]][pos[1]]
            if piece and self._is_my_piece(piece, is_red_turn):
                # 验证是否真的被牵制
                if self._is_pinned(board, pos, pinner_pos, king_pos):
                    results.append(TacticalTagResult(
                        tag_name="is_pinned",
                        detected=True,
                        confidence=1.0,
                        bind_pieces=[
                            self._bind_piece_at(pos, is_red_turn),
                            self._bind_piece_at(pinner_pos, not is_red_turn)
                        ]
                    ))
    
    return results


def detect_cannon_double_attack(self, board: List[List[str]], 
                                 is_red_turn: bool) -> List[TacticalTagResult]:
    """
    检测炮串打标签
    
    算法：
    1. 找到己方炮
    2. 检测是否存在炮架
    3. 检测炮架后是否有两个敌方棋子
    """
    results = []
    
    my_cannons = self._get_my_cannons(board, is_red_turn)
    
    for cannon_pos in my_cannons:
        # 四个方向检测
        for direction in [(0, 1), (0, -1), (1, 0), (-1, 0)]:
            targets = self._find_cannon_targets(board, cannon_pos, direction)
            
            if len(targets) >= 2:
                # 炮串打：同时攻击两个敌方棋子
                results.append(TacticalTagResult(
                    tag_name="is_cannon_double_attack",
                    detected=True,
                    confidence=1.0,
                    bind_pieces=[
                        self._bind_piece_at(cannon_pos, is_red_turn),
                        self._bind_piece_at(targets[0], not is_red_turn),
                        self._bind_piece_at(targets[1], not is_red_turn)
                    ]
                ))
    
    return results
```

#### 2.3.5 第五层：标准杀法检测

```python
def detect_horse_back_cannon(self, board: List[List[str]], 
                              is_red_turn: bool) -> List[TacticalTagResult]:
    """
    检测马后炮杀法
    
    算法：
    1. 找到己方马和炮
    2. 检测马是否控制将/帅的逃跑路线
    3. 检测炮是否将军
    4. 验证是否形成杀棋
    """
    results = []
    
    enemy_king = self._find_king(board, not is_red_turn)
    my_horses = self._get_my_horses(board, is_red_turn)
    my_cannons = self._get_my_cannons(board, is_red_turn)
    
    for horse_pos in my_horses:
        for cannon_pos in my_cannons:
            if self._is_horse_back_cannon(board, horse_pos, cannon_pos, enemy_king):
                results.append(TacticalTagResult(
                    tag_name="checkmate_horse_back_cannon",
                    detected=True,
                    confidence=1.0,
                    bind_pieces=[
                        self._bind_piece_at(horse_pos, is_red_turn),
                        self._bind_piece_at(cannon_pos, is_red_turn),
                        self._bind_piece_at(enemy_king, not is_red_turn)
                    ]
                ))
    
    return results


def detect_iron_gate(self, board: List[List[str]], 
                     is_red_turn: bool) -> List[TacticalTagResult]:
    """
    检测铁门栓杀法
    
    算法：
    1. 找到己方车
    2. 检测车是否在将/帅门线上
    3. 检测是否有其他子力配合封锁
    """
    results = []
    
    enemy_king = self._find_king(board, not is_red_turn)
    my_rooks = self._get_my_rooks(board, is_red_turn)
    
    for rook_pos in my_rooks:
        if self._is_iron_gate(board, rook_pos, enemy_king, is_red_turn):
            results.append(TacticalTagResult(
                tag_name="checkmate_iron_gate",
                detected=True,
                confidence=1.0,
                bind_pieces=[
                    self._bind_piece_at(rook_pos, is_red_turn),
                    self._bind_piece_at(enemy_king, not is_red_turn)
                ]
            ))
    
    return results
```

#### 2.3.6 第六层：子力状态检测

```python
def detect_piece_status(self, board: List[List[str]], 
                        is_red_turn: bool) -> List[TacticalTagResult]:
    """
    检测子力状态层标签
    
    算法：
    1. 遍历所有棋子
    2. 检测是否无保护
    3. 检测炮是否有炮架
    """
    results = []
    
    # 检测所有棋子的保护状态
    for row in range(10):
        for col in range(9):
            piece = board[row][col]
            if not piece:
                continue
            
            is_red = piece.isupper()
            
            # 检测 is_unprotected
            if not self._is_protected(board, (row, col), is_red):
                results.append(TacticalTagResult(
                    tag_name="piece_is_unprotected",
                    detected=True,
                    confidence=1.0,
                    bind_pieces=[self._bind_piece_at((row, col), is_red)]
                ))
            
            # 检测 cannon_has_platform
            if piece.lower() == 'c':
                if self._has_platform(board, (row, col), is_red):
                    results.append(TacticalTagResult(
                        tag_name="cannon_has_platform",
                        detected=True,
                        confidence=1.0,
                        bind_pieces=[self._bind_piece_at((row, col), is_red)]
                    ))
    
    return results
```

### 3.1 模型设计

```python
import torch
import torch.nn as nn
from typing import Dict, Tuple

class XiangqiTransformer(nn.Module):
    """
    象棋Transformer模型
    
    输入：90个棋盘格Token
    输出：
    - 256维高维检索向量
    - 20个战术标签置信度（静态标签）
    - 价值回归（拟合Pikafish score）
    - 策略分类（拟合Pikafish bestmove）
    """
    
    def __init__(
        self,
        d_model: int = 256,
        nhead: int = 8,
        num_layers: int = 6,
        dim_feedforward: int = 1024,
        dropout: float = 0.1,
        num_tags: int = 20,
        num_moves: int = 2086,  # 中国象棋合法走法数
    ):
        super().__init__()
        
        # 棋子嵌入（14类棋子 + 1类空位 = 15类）
        self.piece_embedding = nn.Embedding(15, d_model // 4)
        
        # 位置编码（90个位置）
        self.position_encoding = nn.Parameter(
            torch.randn(90, d_model // 4) * 0.02
        )
        
        # 输入投影
        self.input_proj = nn.Linear(d_model // 2, d_model)
        
        # Transformer编码器
        encoder_layer = nn.TransformerEncoderLayer(
            d_model=d_model,
            nhead=nhead,
            dim_feedforward=dim_feedforward,
            dropout=dropout,
            batch_first=True
        )
        self.transformer = nn.TransformerEncoder(encoder_layer, num_layers)
        
        # 输出头
        self.embedding_head = nn.Linear(d_model, 256)  # 检索向量
        self.tag_head = nn.Linear(d_model, num_tags)    # 战术标签
        self.value_head = nn.Linear(d_model, 1)         # 价值回归
        self.policy_head = nn.Linear(d_model, num_moves) # 策略分类
    
    def forward(
        self, 
        board_tokens: torch.Tensor,
        legal_move_mask: torch.Tensor = None
    ) -> Dict[str, torch.Tensor]:
        """
        前向传播
        
        Args:
            board_tokens: [B, 90] 棋盘Token
            legal_move_mask: [B, num_moves] 合法走法掩码
            
        Returns:
            Dict: {
                "embedding": [B, 256],
                "tags": [B, 20],
                "value": [B, 1],
                "policy": [B, num_moves]
            }
        """
        batch_size = board_tokens.shape[0]
        
        # 棋子嵌入
        piece_emb = self.piece_embedding(board_tokens)  # [B, 90, d_model//4]
        
        # 位置编码
        pos_emb = self.position_encoding.unsqueeze(0).expand(batch_size, -1, -1)
        
        # 拼接
        x = torch.cat([piece_emb, pos_emb], dim=-1)  # [B, 90, d_model//2]
        
        # 输入投影
        x = self.input_proj(x)  # [B, 90, d_model]
        
        # Transformer编码
        x = self.transformer(x)  # [B, 90, d_model]
        
        # 全局池化
        x = x.mean(dim=1)  # [B, d_model]
        
        # 输出头
        outputs = {
            "embedding": self.embedding_head(x),      # [B, 256]
            "tags": torch.sigmoid(self.tag_head(x)),  # [B, 20]
            "value": torch.tanh(self.value_head(x)),  # [B, 1]
            "policy": self.policy_head(x)             # [B, num_moves]
        }
        
        # 应用合法走法掩码
        if legal_move_mask is not None:
            outputs["policy"] = outputs["policy"].masked_fill(
                ~legal_move_mask, float('-inf')
            )
        
        return outputs
```

### 3.3 损失函数设计

```python
class XiangqiLoss(nn.Module):
    """
    多任务损失函数
    
    L = λ1 * BCE(tags_pred, tags_true)      # 战术标签
      + λ2 * MSE(value_pred, value_true)    # 价值回归
      + λ3 * CE(policy_pred, bestmove)      # 策略分类
      + λ4 * ContrastiveLoss(embedding)     # 向量对比学习
    """
    
    def __init__(
        self,
        lambda_tags: float = 1.0,
        lambda_value: float = 0.5,
        lambda_policy: float = 1.0,
        lambda_contrastive: float = 0.1,
    ):
        super().__init__()
        self.lambda_tags = lambda_tags
        self.lambda_value = lambda_value
        self.lambda_policy = lambda_policy
        self.lambda_contrastive = lambda_contrastive
        
        self.bce = nn.BCELoss()
        self.mse = nn.MSELoss()
        self.ce = nn.CrossEntropyLoss()
    
    def forward(
        self,
        outputs: Dict[str, torch.Tensor],
        targets: Dict[str, torch.Tensor],
    ) -> Dict[str, torch.Tensor]:
        """
        计算损失
        
        Args:
            outputs: 模型输出
            targets: 目标值
            
        Returns:
            Dict: {"total": 总损失, "tags": 标签损失, ...}
        """
        # 战术标签损失
        tags_loss = self.bce(outputs["tags"], targets["tags"])
        
        # 价值回归损失
        value_loss = self.mse(outputs["value"], targets["value"])
        
        # 策略分类损失
        policy_loss = self.ce(outputs["policy"], targets["bestmove_idx"])
        
        # 总损失
        total_loss = (
            self.lambda_tags * tags_loss +
            self.lambda_value * value_loss +
            self.lambda_policy * policy_loss
        )
        
        return {
            "total": total_loss,
            "tags": tags_loss,
            "value": value_loss,
            "policy": policy_loss,
        }
```

### 3.4 训练流程

```python
def train_epoch(
    model: XiangqiTransformer,
    dataloader: DataLoader,
    optimizer: torch.optim.Optimizer,
    loss_fn: XiangqiLoss,
    device: str = "cuda",
) -> Dict[str, float]:
    """
    训练一个epoch
    """
    model.train()
    total_loss = 0
    tags_loss = 0
    value_loss = 0
    policy_loss = 0
    
    for batch in dataloader:
        # 移动到设备
        board_tokens = batch.board_tokens.to(device)
        static_tags = batch.static_tags.to(device)
        engine_score = batch.engine_score.to(device)
        bestmove_idx = batch.bestmove_idx.to(device)
        legal_move_mask = batch.legal_move_mask.to(device)
        
        # 前向传播
        outputs = model(board_tokens, legal_move_mask)
        
        # 计算损失
        targets = {
            "tags": static_tags,
            "value": engine_score / 10000.0,  # 归一化到[-1, 1]
            "bestmove_idx": bestmove_idx,
        }
        losses = loss_fn(outputs, targets)
        
        # 反向传播
        optimizer.zero_grad()
        losses["total"].backward()
        optimizer.step()
        
        # 累计损失
        total_loss += losses["total"].item()
        tags_loss += losses["tags"].item()
        value_loss += losses["value"].item()
        policy_loss += losses["policy"].item()
    
    n = len(dataloader)
    return {
        "total": total_loss / n,
        "tags": tags_loss / n,
        "value": value_loss / n,
        "policy": policy_loss / n,
    }
```

### 4.2 数据处理脚本

```python
from pathlib import Path
from typing import List, Dict, Any
import pandas as pd
from tqdm import tqdm

class DataPipeline:
    """
    数据处理流水线
    """
    
    def __init__(
        self,
        pgn_dir: str,
        output_dir: str,
        engine_path: str,
    ):
        self.pgn_dir = Path(pgn_dir)
        self.output_dir = Path(output_dir)
        self.engine_path = engine_path
        
        # 初始化组件
        self.pgn_parser = PGNParser()
        self.engine = PikafishEngine(engine_path)
        self.detector = TacticalDetector(self.engine)
    
    def run(self, num_games: int = None):
        """
        执行数据处理流水线
        """
        all_samples = []
        
        # 1. 解析PGN文件
        pgn_files = list(self.pgn_dir.glob("*.pgn"))
        for pgn_file in tqdm(pgn_files, desc="解析PGN"):
            games = self.pgn_parser.parse(pgn_file)
            
            for game in tqdm(games[:num_games], desc="处理对局", leave=False):
                samples = self._process_game(game)
                all_samples.extend(samples)
        
        # 2. 数据增强
        all_samples = self._augment_data(all_samples)
        
        # 3. 去重
        all_samples = self._deduplicate(all_samples)
        
        # 4. 存储
        self._save(all_samples)
        
        return len(all_samples)
    
    def _process_game(self, game: Dict) -> List[TrainingSample]:
        """
        处理单局对局
        """
        samples = []
        fen = game["initial_fen"]
        moves = game["moves"]
        
        for i, move in enumerate(moves):
            # 获取引擎分析
            engine_result = self.engine.analyze(fen, depth=20)
            
            # 检测静态战术标签
            static_result = self.detector.detect_static(fen)
            
            # 创建训练样本
            sample = TrainingSample(
                fen=fen,
                board_tokens=self._fen_to_tokens(fen),
                static_tags={t.tag_name: t.detected for t in static_result.tags},
                static_tag_pieces={
                    t.tag_name: t.bind_pieces 
                    for t in static_result.tags if t.detected
                },
                engine_bestmove=engine_result.bestmove,
                engine_score=engine_result.score,
                engine_pv=engine_result.pv,
                game_id=game["id"],
                move_number=i // 2 + 1,
                result=game["result"],
            )
            samples.append(sample)
            
            # 执行走法
            fen = self._apply_move(fen, move)
        
        return samples
    
    def _augment_data(self, samples: List[TrainingSample]) -> List[TrainingSample]:
        """
        数据增强
        """
        augmented = []
        
        for sample in samples:
            # 原始样本
            augmented.append(sample)
            
            # 视角归一：转为红方视角
            if self._is_black_turn(sample.fen):
                normalized = self._normalize_to_red_view(sample)
                augmented.append(normalized)
            
            # 左右镜像
            mirrored = self._mirror_horizontal(sample)
            augmented.append(mirrored)
        
        return augmented
    
    def _deduplicate(self, samples: List[TrainingSample]) -> List[TrainingSample]:
        """
        去重
        """
        seen = set()
        unique = []
        
        for sample in samples:
            if sample.fen not in seen:
                seen.add(sample.fen)
                unique.append(sample)
        
        return unique
    
    def _save(self, samples: List[TrainingSample]):
        """
        存储训练数据
        """
        # 转换为DataFrame
        df = pd.DataFrame([{
            "fen": s.fen,
            "board_tokens": s.board_tokens.tolist(),
            "static_tags": s.static_tags,
            "static_tag_pieces": s.static_tag_pieces,
            "engine_bestmove": s.engine_bestmove,
            "engine_score": s.engine_score,
            "engine_pv": s.engine_pv,
            "game_id": s.game_id,
            "move_number": s.move_number,
            "result": s.result,
        } for s in samples])
        
        # 存储为Parquet
        df.to_parquet(self.output_dir / "training_data.parquet")
        
        # 存储为JSONL（便于调试）
        with open(self.output_dir / "training_data.jsonl", "w") as f:
            for s in samples:
                f.write(json.dumps(asdict(s)) + "\n")
```

```python
            }
```
